int MVOFF[7] = { 0, 4, 10, 22, 28, 34, 42 };
int MVCNT[7] = { 4, 6, 12, 6, 6, 8, 10 };
int MFOFF[7] = { 0, 4, 12, 20, 28, 36, 42 };
int MFCNT[7] = { 4, 8, 8, 8, 8, 6, 16 };
int PX[12];
int PY[12];
int EX[12];
int EY[12];

void snapEx() {
    int i = 0;
    while (i < 12) { EX[i] = PX[i];  EY[i] = PY[i];  i = i + 1; }
}

void project(int m, int a, int b) {
    int i, n, base, x, y, z, ca, sa, cb, sb, rx, rz, ry, rz2, zp;
    base = MVOFF[m];
    n = MVCNT[m];
    ca = cos256(a);  sa = sin256(a);
    cb = cos256(b);  sb = sin256(b);
    i = 0;
    while (i < n) {
        x = VX[base + i];  y = VY[base + i];  z = VZ[base + i];
        rx = (x * ca - z * sa) / 256;
        rz = (x * sa + z * ca) / 256;
        ry = (y * cb - rz * sb) / 256;
        rz2 = (y * sb + rz * cb) / 256;
        zp = rz2 + 340;
        PX[i] = 160 + (rx * 256) / zp;
        PY[i] = 132 - (ry * 256) / zp;
        i = i + 1;
    }
}

void wire(int m, int c, int ex) {
    int f, o, n, j, v0, v1, ux, uy, vx, vy, cross;
    f = MFOFF[m];
    while (f < MFOFF[m] + MFCNT[m]) {
        o = FOFF[f];
        n = FLEN[f];
        if (ex == 0) {
            ux = PX[FV[o + 1]] - PX[FV[o]];
            uy = PY[FV[o + 1]] - PY[FV[o]];
            vx = PX[FV[o + 2]] - PX[FV[o]];
            vy = PY[FV[o + 2]] - PY[FV[o]];
            cross = ux * vy - vx * uy;
            if (cross > 0) {
                j = 0;
                while (j < n) {
                    v0 = FV[o + j];
                    v1 = FV[o + (j + 1) % n];
                    line(PX[v0], PY[v0], PX[v1], PY[v1], c);
                    j = j + 1;
                }
            }
        } else {
            j = 0;
            while (j < n) {
                v0 = FV[o + j];
                v1 = FV[o + (j + 1) % n];
                line(EX[v0], EY[v0], EX[v1], EY[v1], 0);
                j = j + 1;
            }
        }
        f = f + 1;
    }
}

int main() {
    int curM, showM, a, b, k;
    init(0);
    curM = 0;  showM = 0;  a = 0;  b = 0;
    project(curM, a, b);
    wire(curM, 3, 0);
    snapEx();
    a = a + 3;  b = b + 1;
    project(curM, a, b);
    while (1) {
        vsync();
        wire(showM, 0, 1);
        wire(curM, 3, 0);
        snapEx();
        k = getkey();
        if (k == 67) { curM = curM + 1;  if (curM > 6) curM = 0; }
        if (k == 68) { curM = curM - 1;  if (curM < 0) curM = 6; }
        a = a + 3;  b = b + 1;
        project(curM, a, b);
        showM = curM;
    }
}
